function init() {
  // Set initial theme
  elements.wizard.setAttribute('data-theme', state.currentTheme);
  
  // Set wizard name
  elements.wizardName.textContent = wizard.name;
  
  // Load saved preferences
  loadPreferences();
  
  // Bind events
  bindEvents();
  
  // Show gesture hint on mobile
  if (isMobile()) {
    setTimeout(() => {
      elements.gestureHint.classList.add('show');
      setTimeout(() => {
        elements.gestureHint.classList.remove('show');
      }, 5000);
    }, 2000);
  }
  
  // Welcome message after delay
  setTimeout(() => {
    showWelcomeMessage();
  }, 3000);
  
  // Start eye tracking
  startEyeTracking();
  
  // Random expression changes
  startExpressionCycle();
}

// ============================================
// EVENT BINDING
// ============================================

function bindEvents() {
  // Wizard click
  elements.wizard.addEventListener('click', handleWizardClick);
  
  // Double click easter egg
  elements.wizard.addEventListener('dblclick', handleDoubleClick);
  
  // Control buttons
  elements.gameModeBtn.addEventListener('click', (e) => {
    e.stopPropagation();
    toggleQuizPanel();
  });
  
  elements.colorBtn.addEventListener('click', (e) => {
    e.stopPropagation();
    toggleColorPanel();
  });
  
  elements.muteBtn.addEventListener('click', (e) => {
    e.stopPropagation();
    toggleMute();
  });
  
  // Panel close buttons
  elements.closeColorPanel.addEventListener('click', toggleColorPanel);
  elements.closeQuiz.addEventListener('click', toggleQuizPanel);
  
  // Color options
  elements.colorOptions.forEach(option => {
    option.addEventListener('click', () => changeTheme(option.dataset.color));
  });
  
  // Quiz buttons
  elements.startQuizBtn.addEventListener('click', startQuiz);
  elements.retryQuizBtn.addEventListener('click', startQuiz);
  
  // Touch gestures
  setupTouchGestures();
  
  // Keyboard shortcuts
  setupKeyboardShortcuts();
  
  // Click outside to close panels
  document.addEventListener('click', handleOutsideClick);
  
  // Voice loading
  window.speechSynthesis.onvoiceschanged = () => {
    console.log('Voices loaded');
  };
}

function handleWizardClick(e) {
  // Don't trigger if clicking on control buttons
  if (e.target.closest('.control-btn')) return;
  
  if (state.isSpeaking) {
    stopSpeaking();
    return;
  }
  
  speak();
}

function handleDoubleClick(e) {
  e.preventDefault();
  
  const secretMessages = [
    "🎉 You found my secret! Here's a virtual high-five!",
    "✨ Double magic! You're a true eco-legend!",
    "🌟 Secret unlocked: You have the heart of a guardian!",
    "🎊 Wow! Double-click power activated!",
    "💫 You've discovered ancient wizard secrets!"
  ];
  
  setExpression('excited');
  showMessage(secretMessages[Math.floor(Math.random() * secretMessages.length)]);
  createSparkles(15);
  playMagicalSound('special');
}

function speak() {
  state.isSpeaking = true;
  
  // Get appropriate message
  const message = getMessage();
  
  // Show message
  showMessage(message);
  
  // Text-to-speech
  if (!state.isMuted) {
    speakWithVoice(message);
  }
  
  // Play sound
  playMagicalSound('speak');
  
  // Increment counter
  state.speechCount++;
  
  // Auto-hide after delay
  setTimeout(() => {
    if (state.isSpeaking) {
      stopSpeaking();
    }
  }, 6000);
}

function stopSpeaking() {
  state.isSpeaking = false;
  elements.speechBubble.classList.remove('active');
  window.speechSynthesis.cancel();
}

function getMessage() {
  const hour = new Date().getHours();
  const random = Math.random();
  
  // First interaction - greeting
  if (state.speechCount === 0) {
    if (hour < 12) {
      return wizard.greetings.morning[Math.floor(Math.random() * wizard.greetings.morning.length)];
    } else if (hour < 17) {
      return wizard.greetings.afternoon[Math.floor(Math.random() * wizard.greetings.afternoon.length)];
    } else if (hour < 21) {
      return wizard.greetings.evening[Math.floor(Math.random() * wizard.greetings.evening.length)];
    } else {
      return wizard.greetings.night[Math.floor(Math.random() * wizard.greetings.night.length)];
    }
  }
  
  // Random message type
  if (random < 0.5) {
    // Eco tip
    setExpression('happy');
    return ecoTips[state.tipIndex % ecoTips.length];
    state.tipIndex++;
  } else if (random < 0.75) {
    // Fun fact
    setExpression('thinking');
    return funFacts[Math.floor(Math.random() * funFacts.length)];
  } else if (random < 0.9) {
    // Motivational quote
    setExpression('love');
    return motivationalQuotes[Math.floor(Math.random() * motivationalQuotes.length)];
  } else {
    // Farewell
    setExpression('wink');
    return wizard.farewells[Math.floor(Math.random() * wizard.farewells.length)];
  }
}

// TEXT-TO-SPEECH
// ============================================

function speakWithVoice(text) {
  window.speechSynthesis.cancel();
  
  const utterance = new SpeechSynthesisUtterance(text);
  
  // Voice settings
  utterance.rate = 0.9;
  utterance.pitch = 0.8;
  utterance.volume = state.isMuted ? 0 : 1;
  
  // Find suitable voice
  const voices = window.speechSynthesis.getVoices();
  const preferredVoice = voices.find(voice => 
    voice.name.includes('Daniel') || 
    voice.name.includes('Google UK English Male') ||
    voice.name.includes('Microsoft David') ||
    voice.name.includes('Male')
  );
  
  if (preferredVoice) {
    utterance.voice = preferredVoice;
  }
  
  // Events
  utterance.onstart = () => {
    animateMouth(true);
  };
  
  utterance.onend = () => {
    animateMouth(false);
  };
  
  utterance.onerror = () => {
    animateMouth(false);
  };
  
  window.speechSynthesis.speak(utterance);
}

function animateMouth(speaking) {
  if (speaking) {
    elements.wizardMouth.style.animation = 'mouthTalk 0.3s infinite';
  } else {
    elements.wizardMouth.style.animation = '';
  }
}

function setExpression(expression) {
  // Remove all expression classes
  elements.wizard.classList.remove('happy', 'excited', 'thinking', 'surprised', 'sleepy', 'love', 'wink', 'angry');
  
  // Add new expression
  if (expression !== 'neutral') {
    elements.wizard.classList.add(expression);
  }
  
  state.currentExpression = expression;
}

function startExpressionCycle() {
  // Change expression based on time
  setInterval(() => {
    if (!state.isSpeaking && !state.quiz.active) {
      const hour = new Date().getHours();
      
      if (hour >= 23 || hour < 6) {
        setExpression('sleepy');
      } else {
        // Random idle expressions
        const idleExpressions = ['happy', 'thinking', 'happy', 'wink'];
        const random = idleExpressions[Math.floor(Math.random() * idleExpressions.length)];
        setExpression(random);
      }
    }
  }, 10000);
}

// ============================================
// EYE TRACKING
// ============================================

function startEyeTracking() {
  document.addEventListener('mousemove', (e) => {
    if (state.isHidden) return;
    
    const wizardRect = elements.wizard.getBoundingClientRect();
    const wizardCenterX = wizardRect.left + wizardRect.width / 2;
    const wizardCenterY = wizardRect.top + wizardRect.height / 2;
    
    const angle = Math.atan2(e.clientY - wizardCenterY, e.clientX - wizardCenterX);
    const distance = Math.min(3, Math.hypot(e.clientX - wizardCenterX, e.clientY - wizardCenterY) / 100);
    
    const pupils = document.querySelectorAll('.pupil');
    pupils.forEach(pupil => {
      pupil.style.transform = `translate(${Math.cos(angle) * distance}px, ${Math.sin(angle) * distance}px)`;
    });
  });
}

function toggleQuizPanel() {
  const isActive = elements.quizPanel.classList.contains('active');
  
  // Close color panel if open
  elements.colorPanel.classList.remove('active');
  
  // Toggle quiz panel
  elements.quizPanel.classList.toggle('active');

    playMagicalSound('open');

    if (!state.quiz.active) {

  } else {

  }

function showQuizScreen(screen) {
  elements.quizStart.style.display = 'none';
  elements.quizGame.style.display = 'none';
  elements.quizResult.style.display = 'none';
  
  switch(screen) {
    case 'start':
      elements.quizStart.style.display = 'block';
      break;
    case 'game':
      elements.quizGame.style.display = 'block';
      break;
    case 'result':
      elements.quizResult.style.display = 'block';
      break;
  }
}

function startQuiz() {
  state.quiz = {
    active: true,
    currentQuestion: 0,
    score: 0,
    answers: []
  };
  
  showQuizScreen('game');
  loadQuestion();
  playMagicalSound('start');
}

function loadQuestion() {
  const question = quizQuestions[state.quiz.currentQuestion];
  
  // Update progress
  const progress = ((state.quiz.currentQuestion) / quizQuestions.length) * 100;
  elements.quizProgressBar.style.width = progress + '%';
  
  // Set question
  elements.quizQuestion.textContent = `Q${state.quiz.currentQuestion + 1}: ${question.question}`;
  
  // Clear and set options
  elements.quizOptions.innerHTML = '';
  question.options.forEach((option, index) => {
    const btn = document.createElement('div');
    btn.className = 'quiz-option';
    btn.textContent = option;
    btn.addEventListener('click', () => selectAnswer(index));
    elements.quizOptions.appendChild(btn);
  });
  
  // Hide feedback
  elements.quizFeedback.className = 'quiz-feedback';
}

function selectAnswer(selectedIndex) {
  const question = quizQuestions[state.quiz.currentQuestion];
  const isCorrect = selectedIndex === question.correct;
  
  // Disable all options
  const options = elements.quizOptions.querySelectorAll('.quiz-option');
  options.forEach((opt, index) => {
    opt.classList.add('disabled');
    if (index === question.correct) {
      opt.classList.add('correct');
    } else if (index === selectedIndex && !isCorrect) {
      opt.classList.add('wrong');
    }
  });
  
  // Show feedback
  elements.quizFeedback.className = 'quiz-feedback show ' + (isCorrect ? 'correct' : 'wrong');
  elements.quizFeedback.textContent = isCorrect 
    ? `✅ Correct! ${question.fact}`
    : `❌ Not quite! ${question.fact}`;
  
  // Update score
  if (isCorrect) {
    state.quiz.score++;
    setExpression('excited');
    playMagicalSound('correct');
    createSparkles(5);
  } else {
    setExpression('thinking');
    playMagicalSound('wrong');
  }
  
  // Move to next question or show results
  setTimeout(() => {
    state.quiz.currentQuestion++;
    
    if (state.quiz.currentQuestion < quizQuestions.length) {
      loadQuestion();
    } else {
      showQuizResults();
    }
  }, 2500);
}

function showQuizResults() {
  state.quiz.active = false;
  
  const score = state.quiz.score;
  const total = quizQuestions.length;
  const percentage = (score / total) * 100;
  
  // Determine badge
  let badge, message;
  
  if (percentage === 100) {
    badge = '🏆';
    message = "Perfect score! You're a true Eco Master! The Earth is lucky to have you!";
    setExpression('love');
  } else if (percentage >= 80) {
    badge = '🥇';
    message = "Excellent! You're an Eco Champion! Keep spreading the green magic!";
    setExpression('excited');
  } else if (percentage >= 60) {
    badge = '🥈';
    message = "Great job! You're an Eco Warrior! A bit more learning and you'll be unstoppable!";
    setExpression('happy');
  } else if (percentage >= 40) {
    badge = '🥉';
    message = "Good effort! You're an Eco Apprentice! Keep learning about our planet!";
    setExpression('thinking');
  } else {
    badge = '📚';
    message = "Every master was once a beginner. Keep exploring eco topics!";
    setExpression('surprised');
  }
  
  // Update UI
  elements.quizBadge.textContent = badge;
  elements.quizScore.textContent = `${score} / ${total}`;
  elements.quizMessage.textContent = message;
  
  // Update progress to 100%
  elements.quizProgressBar.style.width = '100%';
  
  showQuizScreen('result');
  playMagicalSound('complete');
  createSparkles(20);
  
  // Speak results
  if (!state.isMuted) {
    speakWithVoice(`Quiz complete! You scored ${score} out of ${total}. ${message}`);
  }
}

let audioContext = null;

function getAudioContext() {
  if (!audioContext) {
    audioContext = new (window.AudioContext || window.webkitAudioContext)();
  }
  return audioContext;
}

function playMagicalSound(type) {
  if (state.isMuted) return;
  
  try {
    const ctx = getAudioContext();
    const oscillator = ctx.createOscillator();
    const gainNode = ctx.createGain();
    
    oscillator.connect(gainNode);
    gainNode.connect(ctx.destination);
    
    switch(type) {
      case 'speak':
        oscillator.frequency.setValueAtTime(600, ctx.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(800, ctx.currentTime + 0.1);
        gainNode.gain.setValueAtTime(0.1, ctx.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.2);
        oscillator.start(ctx.currentTime);
        oscillator.stop(ctx.currentTime + 0.2);
        break;
        
      case 'correct':
        // Happy ascending notes
        [523, 659, 784].forEach((freq, i) => {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.frequency.value = freq;
          gain.gain.setValueAtTime(0.15, ctx.currentTime + i * 0.1);
          gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + i * 0.1 + 0.3);
          osc.start(ctx.currentTime + i * 0.1);
          osc.stop(ctx.currentTime + i * 0.1 + 0.3);
        });
        break;
        
      case 'wrong':
        oscillator.frequency.setValueAtTime(300, ctx.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(200, ctx.currentTime + 0.3);
        gainNode.gain.setValueAtTime(0.1, ctx.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.3);
        oscillator.start(ctx.currentTime);
        oscillator.stop(ctx.currentTime + 0.3);
        break;
        
      case 'special':
        // Magical arpeggio
        [523, 659, 784, 1047, 784, 659, 523].forEach((freq, i) => {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.frequency.value = freq;
          gain.gain.setValueAtTime(0.1, ctx.currentTime + i * 0.08);
          gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + i * 0.08 + 0.2);
          osc.start(ctx.currentTime + i * 0.08);
          osc.stop(ctx.currentTime + i * 0.08 + 0.2);
        });
        break;
        
      case 'theme':
        oscillator.frequency.setValueAtTime(400, ctx.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(800, ctx.currentTime + 0.3);
        gainNode.gain.setValueAtTime(0.1, ctx.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.3);
        oscillator.start(ctx.currentTime);
        oscillator.stop(ctx.currentTime + 0.3);
        break;
        
      case 'open':
        oscillator.frequency.setValueAtTime(500, ctx.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(700, ctx.currentTime + 0.15);
        gainNode.gain.setValueAtTime(0.08, ctx.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.15);
        oscillator.start(ctx.currentTime);
        oscillator.stop(ctx.currentTime + 0.15);
        break;
        
      case 'start':
        [440, 554, 659].forEach((freq, i) => {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.frequency.value = freq;
          gain.gain.setValueAtTime(0.1, ctx.currentTime + i * 0.1);
          gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + i * 0.1 + 0.2);
          osc.start(ctx.currentTime + i * 0.1);
          osc.stop(ctx.currentTime + i * 0.1 + 0.2);
        });
        break;
        
      case 'complete':
        // Fanfare
        [523, 523, 523, 659, 784, 784, 1047].forEach((freq, i) => {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.frequency.value = freq;
          const delay = i < 3 ? i * 0.15 : 0.45 + (i - 3) * 0.2;
          gain.gain.setValueAtTime(0.12, ctx.currentTime + delay);
          gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + delay + 0.3);
          osc.start(ctx.currentTime + delay);
          osc.stop(ctx.currentTime + delay + 0.3);
        });
        break;
    }
  } catch (e) {
    console.log('Audio not supported');
  }
}

function createSparkles(count = 10) {
  const wizardRect = elements.wizard.getBoundingClientRect();
  const centerX = wizardRect.left + wizardRect.width / 2;
  const centerY = wizardRect.top + wizardRect.height / 2;
  
  for (let i = 0; i < count; i++) {
    const sparkle = document.createElement('div');
    sparkle.className = 'sparkle';
    
    const size = 5 + Math.random() * 10;
    sparkle.style.width = size + 'px';
    sparkle.style.height = size + 'px';
    sparkle.style.left = centerX + 'px';
    sparkle.style.top = centerY + 'px';
    
    document.body.appendChild(sparkle);
    
    // Animate
    const angle = (Math.PI * 2 * i) / count + Math.random() * 0.5;
    const velocity = 3 + Math.random() * 5;
    let vx = Math.cos(angle) * velocity;
    let vy = Math.sin(angle) * velocity - 3;
    let life = 1;
    let rotation = 0;
    
    const animate = () => {
      life -= 0.015;
      rotation += 5;
      vy += 0.2; // gravity
      
      sparkle.style.left = parseFloat(sparkle.style.left) + vx + 'px';
      sparkle.style.top = parseFloat(sparkle.style.top) + vy + 'px';
      sparkle.style.opacity = life;
      sparkle.style.transform = `scale(${life}) rotate(${rotation}deg)`;
      
      if (life > 0) {
        requestAnimationFrame(animate);
      } else {
        sparkle.remove();
      }
    };
    
    requestAnimationFrame(animate);
  }
}

function setupTouchGestures() {
  let touchStartX = 0;
  let touchStartY = 0;
  let touchEndX = 0;
  let touchEndY = 0;
  
  elements.wizard.addEventListener('touchstart', (e) => {
    touchStartX = e.changedTouches[0].screenX;
    touchStartY = e.changedTouches[0].screenY;
  }, { passive: true });
  
  elements.wizard.addEventListener('touchend', (e) => {
    touchEndX = e.changedTouches[0].screenX;
    touchEndY = e.changedTouches[0].screenY;
    handleGesture();
  }, { passive: true });
  
  function handleGesture() {
    const diffX = touchEndX - touchStartX;
    const diffY = touchEndY - touchStartY;
    const threshold = 50;
    
    if (Math.abs(diffX) > Math.abs(diffY)) {
      // Horizontal swipe
      if (diffX > threshold) {
        // Swipe right - Open colors
        toggleColorPanel();
        showGestureFeedback('Colors opened! 🎨');
      } else if (diffX < -threshold) {
        // Swipe left - Open quiz
        toggleQuizPanel();
        showGestureFeedback('Quiz opened! 🎮');
      }
    } else {
      // Vertical swipe
      if (diffY > threshold) {
        // Swipe down - Hide wizard
        toggleWizardVisibility();
        showGestureFeedback('Wizard hidden! 👋');
      } else if (diffY < -threshold) {
        // Swipe up - Show wizard
        if (state.isHidden) {
          toggleWizardVisibility();
          showGestureFeedback('Wizard is back! ✨');
        }
      }
    }
  }
}

function showGestureFeedback(message) {
  const feedback = document.createElement('div');
  feedback.style.cssText = `
    position: fixed;
    bottom: 200px;
    right: 30px;
    background: rgba(0, 0, 0, 0.8);
    color: white;
    padding: 10px 20px;
    border-radius: 20px;
    font-size: 0.9rem;
    z-index: 1003;
    animation: fadeInOut 2s ease forwards;
  `;
  feedback.textContent = message;
  document.body.appendChild(feedback);
  
  setTimeout(() => feedback.remove(), 2000);
}

function setupKeyboardShortcuts() {
  document.addEventListener('keydown', (e) => {
    // Only if wizard is not hidden
    if (state.isHidden) return;
    
    switch(e.key.toLowerCase()) {
      case 'w':
        // Trigger wizard speak
        if (!state.isSpeaking) speak();
        break;
      case 'q':
        // Toggle quiz
        toggleQuizPanel();
        break;
      case 'c':
        // Toggle colors
        toggleColorPanel();
        break;
      case 'm':
        // Toggle mute
        toggleMute();
        break;
      case 'h':
        // Toggle hide
        toggleWizardVisibility();
        break;
    }
  });
}

function typeText(element, text, speed = 30) {
  element.textContent = '';
  let i = 0;
  
  const timer = setInterval(() => {
    element.textContent += text[i];
    i++;
    
    if (i >= text.length) {
      clearInterval(timer);
    }
  }, speed);
  
  // Store timer to cancel if needed
  element._typeTimer = timer;
}

function toggleMute() {
  state.isMuted = !state.isMuted;
  elements.muteBtn.textContent = state.isMuted ? '🔇' : '🔊';
  
  if (state.isMuted) {
    window.speechSynthesis.cancel();
    showGestureFeedback('Muted 🔇');
  } else {
    showGestureFeedback('Sound on 🔊');
  }
  
  savePreferences();
}

function toggleWizardVisibility() {
  state.isHidden = !state.isHidden;
  elements.wizard.classList.toggle('wizard-hidden', state.isHidden);
  
  if (state.isHidden) {
    // Close all panels
    elements.speechBubble.classList.remove('active');
    elements.colorPanel.classList.remove('active');
    elements.quizPanel.classList.remove('active');
    window.speechSynthesis.cancel();
  }
}

function handleOutsideClick(e) {
  // Close panels if clicking outside
  if (!e.target.closest('.eco-wizard') && 
      !e.target.closest('.color-panel') && 
      !e.target.closest('.quiz-panel')) {
    elements.colorPanel.classList.remove('active');
    elements.quizPanel.classList.remove('active');
  }
}

function isMobile() {
  return /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
}

function savePreferences() {
  const prefs = {
    theme: state.currentTheme,
    muted: state.isMuted,
    hidden: state.isHidden
  };
  
  try {
    localStorage.setItem('ecoWizardPrefs', JSON.stringify(prefs));
  } catch (e) {
    console.log('Could not save preferences');
  }
}

function loadPreferences() {
  try {
    const saved = localStorage.getItem('ecoWizardPrefs');
    if (saved) {
      const prefs = JSON.parse(saved);
      
      if (prefs.theme) {
        state.currentTheme = prefs.theme;
        elements.wizard.setAttribute('data-theme', prefs.theme);
        
        // Update selected color
        elements.colorOptions.forEach(opt => {
          opt.classList.toggle('selected', opt.dataset.color === prefs.theme);
        });
      }
      
      if (prefs.muted) {
        state.isMuted = true;
        elements.muteBtn.textContent = '🔇';
      }
      
      if (prefs.hidden) {
        state.isHidden = true;
        elements.wizard.classList.add('wizard-hidden');
      }
    }
  } catch (e) {
    console.log('Could not load preferences');
  }
}
