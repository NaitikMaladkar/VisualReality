   254→  // Strip emoji / pictographs / subscripts so TTS reads clean prose
   255→  function speakableText(text) {

   258→        /[\u{1F000}-\u{1FAFF}\u{2190}-\u{21FF}\u{2600}-\u{27BF}\u{2B00}-\u{2BFF}\u{FE0F}\u{200D}]/gu,

   261→      .replace(/[\u2080-\u209C]/g, function (ch) {
   262→        return String(ch.charCodeAt(0) - 0x2080);
   263→      })
   264→      .replace(/\s*([.,!?;:])/g, "$1")
   265→      .replace(/\s+/g, " ")
   266→      .trim();
   267→  }

   367→  function getAudioContext() {
   368→    var Ctx = window.AudioContext || window.webkitAudioContext;
   369→    if (!Ctx) return null;
   370→    if (!audioCtx) audioCtx = new Ctx();

   373→    return audioCtx;

   376→  function tone(ctx, freq, start, dur, volume) {
   377→    var osc = ctx.createOscillator();
   378→    var gain = ctx.createGain();
   379→    osc.connect(gain);
   380→    gain.connect(ctx.destination);
   381→    osc.frequency.value = freq;
   382→    gain.gain.setValueAtTime(volume, ctx.currentTime + start);
   383→    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + start + dur);
   384→    osc.start(ctx.currentTime + start);
   385→    osc.stop(ctx.currentTime + start + dur);
   386→  }

   441→  function createSparkles(count) {
   442→    if (reduceMotion) return;
   443→    var rect = el.wizard.getBoundingClientRect();
   444→    var centerX = rect.left + rect.width / 2;
   445→    var centerY = rect.top + rect.height / 2;
   446→
   447→    for (var i = 0; i < count; i++) {
   448→      (function () {
   449→        var sparkle = document.createElement("div");
   450→        sparkle.className = "sparkle";
   451→        var size = 5 + Math.random() * 9;
   452→        sparkle.style.width = size + "px";
   453→        sparkle.style.height = size + "px";
   454→        sparkle.style.left = centerX + "px";
   455→        sparkle.style.top = centerY + "px";
   456→        document.body.appendChild(sparkle);
   457→
   458→        var angle = (Math.PI * 2 * i) / count + Math.random() * 0.5;
   459→        var velocity = 3 + Math.random() * 5;
   460→        var vx = Math.cos(angle) * velocity;
   461→        var vy = Math.sin(angle) * velocity - 3;
   462→        var life = 1;
   463→        var rotation = 0;
   464→
   465→        function step() {
   466→          life -= 0.015;
   467→          rotation += 5;
   468→          vy += 0.2;
   469→          sparkle.style.left = parseFloat(sparkle.style.left) + vx + "px";
   470→          sparkle.style.top = parseFloat(sparkle.style.top) + vy + "px";
   471→          sparkle.style.opacity = Math.max(life, 0);
   472→          sparkle.style.transform =
   473→            "scale(" + Math.max(life, 0) + ") rotate(" + rotation + "deg)";
   474→          if (life > 0) {
   475→            window.requestAnimationFrame(step);
   476→          } else {
   477→            sparkle.remove();
   478→          }
   479→        }
   480→        window.requestAnimationFrame(step);
   481→      })();
   482→    }
   483→  }

   609→      var rect = el.wizard.getBoundingClientRect();
   610→      var cx = rect.left + rect.width / 2;
   611→      var cy = rect.top + rect.height / 2;
   612→      var angle = Math.atan2(e.clientY - cy, e.clientX - cx);
   613→      var distance = Math.min(3, Math.hypot(e.clientX - cx, e.clientY - cy) / 100);
   614→
   615→      var pupils = el.wizard.querySelectorAll(".pupil");
   616→      Array.prototype.forEach.call(pupils, function (pupil) {
   617→        pupil.style.transform =
   618→          "translate(" +
   619→          Math.cos(angle) * distance +
   620→          "px," +
   621→          Math.sin(angle) * distance +
   622→          "px)";
   623→      });
   624→    });
   625→  }

   935→  function handleKeyDown(e) {
   936→    if (e.metaKey || e.ctrlKey || e.altKey) return;
   937→    if (
   938→      e.target.closest &&
   939→      e.target.closest("input, textarea, select, [contenteditable]")
   940→    ) {
   941→      return;
   942→    }
   943→
   944→    switch (e.key.toLowerCase()) {
   945→      case "w":
   946→        if (state.isSpeaking) stopSpeaking();
   947→        else if (!state.isHidden) speak();
   948→        break;
   949→      case "q":
   950→        if (!state.isHidden) toggleQuizPanel();
   951→        break;
   952→      case "c":
   953→        if (!state.isHidden) toggleColorPanel();
   954→        break;
   955→      case "m":
   956→        toggleMute();
   957→        break;
   958→      case "h":
   959→        toggleWizardVisibility(); // works both ways — fixes the dead-end
   960→        break;
   961→      case "escape":
   962→        el.colorPanel.classList.remove("active");
   963→        el.quizPanel.classList.remove("active");
   964→        if (state.isSpeaking) stopSpeaking();
   965→        break;
   966→    }
   967→  }

   980→  function setupTouchGestures() {
   981→    var startX = 0, startY = 0;
   982→
   983→    el.wizard.addEventListener(
   984→      "touchstart",
   985→      function (e) {
   986→        startX = e.changedTouches[0].screenX;
   987→        startY = e.changedTouches[0].screenY;
   988→      },
   989→      { passive: true }
   990→    );
   991→
   992→    el.wizard.addEventListener(
   993→      "touchend",
   994→      function (e) {
   995→        var diffX = e.changedTouches[0].screenX - startX;
   996→        var diffY = e.changedTouches[0].screenY - startY;
   997→        var threshold = 50;
   998→
   999→        if (Math.abs(diffX) <= threshold && Math.abs(diffY) <= threshold) return;
  1000→
  1001→        if (Math.abs(diffX) > Math.abs(diffY)) {
  1002→          if (diffX > 0) {
  1003→            toggleColorPanel();
  1004→            showToast("Colors 🎨");
  1005→          } else {
  1006→            toggleQuizPanel();
  1007→            showToast("Quiz 🎮");
  1008→          }
  1009→        } else if (diffY > 0) {
  1010→          toggleWizardVisibility(true);
  1011→          showToast("Wizard hidden 👋");
  1012→        }
  1013→      },
  1014→      { passive: true }
  1015→    );
  1016→  }
